117{
118
119 DesignTreeItemData* data = ( DesignTreeItemData* ) this->GetItemData( treeID );
120 Design::AlbumBaseType nodeType = data->GetType( );

134 if ( node )

140 wxString albumCatCodes = albumStamp->GetCatalogCodes( );

158 wxTreeItemIdValue cookie;

171// AddChild adds wxXmlNode as a item in the tree. It is recursively called to
172// create sort nodes as necessary to find the proper place for the child
173wxTreeItemId AlbumTreeCtrl::AddChild( wxTreeItemId parent, wxXmlNode* child )
174{
175 wxString name = child->GetName( );
176 Design::AlbumBaseType nodeType = Design::FindAlbumBaseType( name );
177
178 if ( Design::IsAlbumBaseTypeValid( nodeType ) )
179 {
180 Design::AlbumBase* node = Design::MakeNode( child );
181 if ( node )
182 {
183 node->LoadFonts( child );
184
185 Design::IconID icon;
186 wxString label;
187
188 //if the child element is not a stamp
189 if ( nodeType == Design::AT_Stamp )
190 {
191 // then we add the appropriate icon and label
192 Design::Stamp* stamp = ( Design::Stamp* ) node;
193 // stamp combines the stampID and its name to form a label
194
195 label = stamp->GetCatalogNbr( );
196 wxString title = stamp->GetAttrStr( Design::AT_Name );
197 label += " - " + title;
198 icon = GetIconId( stamp );
199 }
200 else
201 {
202 if ( nodeType == Design::AT_Album )
203 {
204 GetAlbumVolume( )->SetAlbum( ( Design::Album* ) node );
205 ( ( Design::Album* ) node )->FixupNode( );
206 ( ( Design::Album* ) node )->LoadDefaults( child );
207 }
208 label = name;
209 wxString title = node->GetAttrStr( Design::AT_Name );
210 label += " - " + title;
211 icon = Design::Icon_Folder;
212 }
213
214
215 // create the item data and add it to the tree
216 DesignTreeItemData* itemData = new DesignTreeItemData( nodeType, "", node );
217 wxTreeItemId childID;
218 if ( parent.IsOk( ) )
219 {
220 childID = AppendItem( parent, label, icon, -1, itemData );
221 }
222 else
223 {
224 childID = AddRoot( label, icon, -1, itemData );
225 }
226 node->SetTreeItemId( childID );
227
228
229 if ( nodeType == Design::AT_Stamp )
230 {
231 Utils::StampList* stampList = GetStampAlbumCatalogLink( );
232 wxString nodeId = node->GetAttrStr( Design::AT_CatNbr );
233 if ( !nodeId.IsEmpty( ) )
234 {
235 Utils::StampLink* link = stampList->AddStamp( nodeId );
236 link->SetDesignTreeID( childID );
237 SetItemStampLink( childID, link );
238 }
239 //set the icon for the appropriate state
240 SetItemImage( childID, icon );
241 // SetItemState( childID, 0 );
242 }
243 else
244 {
245 // not a stamp just add the appropriate image
246 SetItemStampLink( childID, 0 );
247 SetItemImage( childID, Design::Icon_Folder );
248 // SetItemState( childID, wxTREE_ITEMSTATE_NEXT );
249 }
250
251 // now do it all again for this stamps children
252 wxXmlNode* grandChild = child->GetChildren( );
253 while ( grandChild )
254 {
255 // start adding child elements to it.
256 AddChild( childID, grandChild );
257 grandChild = grandChild->GetNext( );
258 }
259 return childID;
260 }
261 }
262 return ( wxTreeItemId ) 0;
263}

353{
354
355 Utils::StampLink* newLink = ( Utils::StampLink* ) 0;
356 wxTreeItemId currAlbumID = GetSelection( );
357 if ( currAlbumID.IsOk( ) && catTreeID.IsOk( ) )
358 {
359
360 // A stamp can't have children. so get its parent
361 Design::AlbumBaseType type = GetItemType( currAlbumID );
362 if ( type == Design::AT_Stamp ) //@@@ || type == Design::AT_Title )
363 {
364 currAlbumID = GetItemParent( currAlbumID );
365 }
366
367 Design::Stamp* newStamp = CreateNewStamp( catTreeID );
368
369 if ( newStamp )
370 {
371 // gather info from the catalog item entry and update thenew stamp
372 wxString catStr = GetCatalogData( )->GetCatalogVolume( )->GetFilename( );
373 wxFileName catFileName( catStr );
374
375
376 CatalogTreeCtrl* catTree = GetCatalogTreeCtrl( );
377 wxString idText = catTree->GetID( catTreeID );
378 wxString str = newStamp->MakeDisplayNbr( );
379
380 newStamp->GetNbrFrame( )->SetString( str );
381 newStamp->GetNameFrame( )->SetString( newStamp->GetAttrStr( Design::AT_Name ) );
382
383
384 AddStampTreeItem( currAlbumID, newStamp );
385 wxTreeItemId newStampID = newStamp->GetTreeItemId( );
386
387 SetItemText( newStampID, idText );
388 newStamp->SetAttrStr( Design::AT_CatNbr, idText );
389
390 // stamp item text combines the stampID and its name to form a label
391 wxString label = newStamp->GetAttrStr( Design::AT_CatNbr );
392 label += " - " + newStamp->GetAttrStr( Design::AT_Name );
393 SetItemText( newStampID, label );
394
395 //Add the StampList link
396 Utils::StampList* stampList = GetStampAlbumCatalogLink( );
397 newLink = stampList->AddStamp( idText );
398 wxTreeItemId albumID = newStamp->GetTreeItemId( );
399 SetItemStampLink( albumID, newLink );
400 newLink->SetDesignTreeID( albumID );
401 newLink->SetCatTreeID( catTreeID );
402
403 UpdateItemPageLayout( albumID );
404
405 Design::NodeStatus status = newStamp->ValidateNode( );
406 SetItemBackgroundColour( albumID, ItemBackgroundColour[ status ] );
407 }
408 }
409 return newLink;

435{
436 if ( !dstID.IsOk( ) || !srcID.IsOk( ) )
437 {
438 return;
439 }
440 DesignTreeItemData* data = ( DesignTreeItemData* ) GetItemData( srcID );
441 DesignTreeItemData* itemData = new DesignTreeItemData( *data );
442 Design::AlbumBaseType srcType = AlbumTreeCtrl::GetItemType( srcID );
443 Design::AlbumBaseType destType = AlbumTreeCtrl::GetItemType( dstID );
444 wxTreeItemId id;
445 if ( srcType == destType )
446 {
447 wxTreeItemId parent = GetItemParent( dstID );
448 id = InsertItem( parent, dstID,
449 GetItemText( srcID ),
450 GetItemImage( srcID ),
451 -1,
452 itemData );
453 }
454 else
455 {
456 id = AppendItem( dstID,
457 GetItemText( srcID ),
458 GetItemImage( srcID ),
459 -1,
460 itemData );
461 }
462 wxTreeItemIdValue cookie;
463 wxTreeItemId child = GetFirstChild( srcID, cookie );
464 Design::AlbumBase* node = GetItemNode( id );
465 if ( node )
466 {
467 node->SetTreeItemId( id );
468 }
469 Utils::StampLink* link = GetItemStampLink( id );
470 if ( link )
471 {
472 link->SetDesignTreeID( id );
473 }
474 while ( child )
475 {
476 CopyItem( id, child );
477 child = GetNextChild( srcID, cookie );
478 }
479}
