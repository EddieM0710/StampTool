186 bool CSVData::FixUpLine( wxString& line, int lineNbr )
187 {
188 int curr = 0;
189 int last = line.length( );
190 int firstQuote = 0;
191 int nextQuote = 0;
192 int comma = 0;
193 while ( curr < last )
194 {
195 firstQuote = line.find( "\"", curr );
196 if ( firstQuote < 0 )
197 {
198 return true;
199 }
200 nextQuote = line.find_first_of( "\"", firstQuote + 1 );
201 if ( nextQuote < 0 )
202 {
203 // unmatched parens
204 // this means the csv is messed up
205 wxString caption = "CSV Format Error";
206 wxString message;
207 message = message.Format( "Format error in csv file %s at line number %d. \n\n%s", m_filename, lineNbr, line );
208 wxMessageDialog* msg = new wxMessageDialog( GetFrame( ),
209 message, caption, wxOK | wxCENTRE | wxICON_ERROR );
210 msg->ShowModal( );
211 msg->~wxMessageDialog( );
212 return false;
213 }
214 comma = line.find_first_of( ",", firstQuote );
215 bool check_again = true;
216 while ( check_again )
217 {
218 if ( ( comma > firstQuote ) && ( comma < nextQuote ) )
219 {
220 line = line.replace( comma, 1, "{" );
221 comma = line.find_first_of( ",", firstQuote );
222 }
223 else
224 {
225 check_again = false;
226 }
227 }
228 curr = nextQuote + 1;
229 }
230 return true;
231 };
