55 CatalogCode::CatalogCode( wxString codeList )
56 {
57 m_codeList = codeList;
58 // m_codes.Empty( );
59 m_codes.Clear( );
60 wxString delim = ",";
61 if ( !m_codeList.IsEmpty( ) )
62 {
63 int cnt = 0;
64 wxStringTokenizer tokenizer( m_codeList, delim, wxTOKEN_DEFAULT );
65 while ( tokenizer.HasMoreTokens( ) )
66 {
67 wxString token = tokenizer.GetNextToken( );
68 m_codes.Add( token );
69 m_codes[ cnt ].Trim( true );
70 m_codes[ cnt ].Trim( false );
71 cnt++;
72 }
73 }
