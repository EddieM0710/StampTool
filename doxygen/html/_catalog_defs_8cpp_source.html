339 wxXmlNode* InsertEntry( wxXmlNode* sibling, wxXmlNode* child, bool after )
340 {
341 if ( sibling == child )
342 {
343 // can't be a child of itself
344 return ( wxXmlNode* ) 0;
345 }
346
347 // Make a copy of the old child and insert it
348 wxXmlNode* newChildNode = new wxXmlNode( *child );
349 wxXmlNode* newChild = newChildNode;
350 wxXmlNode* parent = sibling->GetParent( );
351
352 if ( after )
353 {
354 if ( !parent->InsertChildAfter( newChild, sibling ) )
355 {
356 return ( wxXmlNode* ) 0;;
357 }
358 }
359 else
360 {
361 if ( !parent->InsertChild( newChild, sibling ) )
362 {
363 return ( wxXmlNode* ) 0;;
364 }
365 }
366
367 wxXmlNode* previousParent = child->GetParent( );
368 previousParent->RemoveChild( child );
369 return newChild;
370 }

372 wxXmlNode* MoveEntry( wxXmlNode* newParent, wxXmlNode* child )
373 {
374 if ( newParent == child )
375 {
376 // can't be a child of itself
377 return ( wxXmlNode* ) 0;
378 }
379
380 wxXmlNode* parent = newParent->GetParent( );
381 while ( parent )
382 {
383 if ( child == parent )
384 {
385 // child can't be an ancestor of itself
386 return ( wxXmlNode* ) 0;
387 }
388 parent = parent->GetParent( );
389 }
390 // Make a copy of the old child and insert it
391 wxXmlNode* newChildNode = new wxXmlNode( *child );
392 wxXmlNode* newChild = newChildNode;
393 newParent->AddChild( newChild );
394
395 parent = child->GetParent( );
396 parent->RemoveChild( child );
397 return newChild;
398 }
