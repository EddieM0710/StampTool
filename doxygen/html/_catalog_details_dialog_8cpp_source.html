191void CatalogDetailsDialog::OnCatDirClick( wxCommandEvent& event )
192{
193 wxString catfile = m_catPath->GetValue( );
194 wxFileName filename1( catfile );
195 wxString path = filename1.GetPath( );
196 wxString name = filename1.GetFullName( );
197 wxFileDialog dirDialog( this, "Select the art directory", path, name,
198 "Catalog files (*.cat)|*.cat",
199 wxFD_SAVE );
200
201 if ( dirDialog.ShowModal( ) == wxID_CANCEL )
202 return;
203
204 wxString filename = dirDialog.GetPath( );
205 m_catPath->SetValue( filename );
206
207}
