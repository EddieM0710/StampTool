173void CatalogTOCTreeCtrl::OnRightDClick( wxMouseEvent& event )
174{
175 wxTreeItemId itemId = GetSelection( );

178 if ( type == Utils::TOC_Volume )

183 GetCatalogData( )->GetCatalogPanel( )->SetVolumeListCtrl( );
184

186
187 if ( generateListPanel == ( GenerateList* ) GetFrame( )->GetStampToolPanel( )->GetPage( ) )

192 {
