215 bool Column::UpdateMinimumSize( )
216 {
217 //Positioning down the col.
218 //The min width of the col is the size of the widest child
219 //The min height is the sum of the min heights of the children.
220 //plus the title if used
221 double minWidth = 0.0;
222 double minHeight = 0.0;
223
224 wxTreeItemIdValue cookie;
225 wxTreeItemId parentID = GetTreeItemId( );
226 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
227 while ( childID.IsOk( ) )
228 {
229 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
230
231 // calc the min size ror this items children
232 child->UpdateMinimumSize( );
233
234 // keep the max min width
235
236 if ( child->GetMinWidth( ) > minWidth )
237 {
238 minWidth = child->GetMinWidth( );
239 }
240
241 // keep running total of min child height
242 minHeight += child->GetMinHeight( );
243 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
244 }
245
246 GetTitleFrame( )->Init( );
247 if ( GetShowTitle( ) )
248 {
249 // update the title frame
250 m_titleFrame->UpdateString( minWidth, minWidth );
251
252 //***
253 // SetHeight( GetHeight( ) + GetTitleFrame( )->GetHeight( ) );
254
255 // add the title height
256 minHeight += GetTitleFrame( )->GetHeight( );
257
258 GetTitleFrame( )->SetYPos( 0 );
259 }
260
261 SetMinHeight( minHeight
262 + GetTopContentMargin( )
263 + GetBottomContentMargin( ) );
264 SetMinWidth( minWidth
265 + GetLeftContentMargin( )
266 + GetRightContentMargin( ) );
267
268 GetErrorArray( )->Empty( );
269
270 return true;
271 }

275 void Column::UpdatePositions( )
276 {
277 // go to the bottom of each child container object ( row, column, page )
278 // and begin filling in position relative to the parent
279
280 // calc space used by children
281 // this means looping through the children and getting the actual sizes
282 // of the children, not their calculated min size
283 wxTreeItemIdValue cookie;
284 wxTreeItemId parentID = GetTreeItemId( );
285 LayoutBase* parent = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( parentID );
286 int nbrChildren = parent->GetNbrChildren( );
287
288 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
289 double actualWidth = 0;
290 double actualHeight = 0;
291 while ( childID.IsOk( ) )
292 {
293 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
294 actualHeight += child->GetHeight( );
295 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
296 }
297
298 double xPos = 0;
299 double yPos = GetTopContentMargin( );//0;
300 //First calc title position
301 if ( GetShowTitle( ) )
302 {
303 GetTitleFrame( )->SetXPos( ( GetWidth( ) - GetTitleFrame( )->GetWidth( ) ) / 2 );
304 GetTitleFrame( )->SetYPos( 2 );
305 // allow for space above title, title height and that much again for nice spaing
306 yPos += GetTitleFrame( )->GetHeight( ) + 2 * GetTopContentMargin( );
307 }
308 else
309 {
310 GetTitleFrame( )->Init( );
311 yPos = GetTopContentMargin( );
312 }
313
314 // this is a col so we are positioning children down the page
315 //double spacing = ( GetHeight( ) - GetMinHeight( ) ) / ( nbrRows + nbrStamps + 1 );
316 double spacing = 4;
317 if ( CalculateSpacing( ) )
318 {
319 spacing = ( GetHeight( ) - actualHeight ) / ( nbrChildren + 1 );
320
321 // inital x/y pos within the row
322 yPos += spacing;
323 }
324 else
325 {
326 spacing = GetFixedSpacingDbl( );
327 yPos += ( ( GetHeight( )
328 - actualHeight )
329 - ( nbrChildren - 1 ) * spacing ) / 2;
330 }
331
332 childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
333 while ( childID.IsOk( ) )
334 {
335 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
336 child->SetXPos( xPos );
337 child->SetYPos( yPos );
338
339
340 child->UpdatePositions( );
341
342 yPos += child->GetHeight( ) + spacing;
343
344 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
345 }
346
347 ValidateNode( );
348 }

352 void Column::UpdateSizes( )
353 {
354 // Set the height and width of each child column
355 // Stamps have fixed height and width
356 wxTreeItemIdValue cookie;
357 wxTreeItemId parentID = GetTreeItemId( );
358 LayoutBase* parent = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( parentID );
359 int nbrChildren = parent->GetNbrChildren( );
360
361 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
362 while ( childID.IsOk( ) )
363 {
364 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
365 if ( child->IsNodeType( AT_Row ) ) // i.e., stamps are fixed don't try to change them
366 {
367 child->SetWidth( GetWidth( )
368 - GetLeftContentMargin( )
369 - GetRightContentMargin( ) );
370 child->SetHeight( child->GetMinHeight( )
371 + ( GetHeight( ) - GetMinHeight( ) ) / ( 2 * nbrChildren )
372 - GetLeftContentMargin( )
373 - GetRightContentMargin( ) );
374 }
375 child->UpdateSizes( );
376
377 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
378 }
