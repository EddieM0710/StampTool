187void CompEntry::Update( Catalog::MergeData* mergeEntry, Catalog::DataTypes type )
188{
189 m_type = type;
190 m_targetFileEntry = mergeEntry->GetTargetEntry( );
191 m_mergeFileEntry = mergeEntry->GetMergeEntry( );
192 if ( m_targetFileEntry )
193 {
194 SetTargetValue( m_targetFileEntry->GetAttr( m_type ) );
195 m_merge->Enable( );
196 m_undo->Enable( );
197 }
198 else
199 {
200 SetTargetValue( "" );
201 m_merge->Disable( );
202 m_undo->Disable( );
203 }
204 if ( m_mergeFileEntry )
205 {
206 SetMergeValue( m_mergeFileEntry->GetAttr( m_type ) );
207 m_merge->Enable( );
208 m_undo->Enable( );
209 }
210 else
211 {
212 SetMergeValue( "" );
213 m_merge->Disable( );
214 m_undo->Disable( );
215 }
216
217 SetBackground( );
218}
