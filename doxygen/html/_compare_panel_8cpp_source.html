533void ComparePanel::OnCopyAllClick( wxCommandEvent& event )
534{
535 // long sel = m_listBox->GetFocusedItem( );
536 wxTreeItemId sel = m_compareTreeCtrl->GetSelection( );
537
538
539 CompareTreeItemData* data = ( CompareTreeItemData* ) m_compareTreeCtrl->GetItemData( sel );
540 Catalog::MergeData* mergeEntry = ( Catalog::MergeData* ) data->GetMergeData( );;
541 Catalog::Entry* targetCatEntry = mergeEntry->GetTargetEntry( );
542 Catalog::Entry* mergeCatEntry = mergeEntry->GetMergeEntry( );
543 if ( !targetCatEntry )
544 {
545 wxXmlNode* node = mergeCatEntry->GetCatXMLNode( );
546
547 wxXmlNode* newTargetNode = new wxXmlNode( 0, wxXML_ELEMENT_NODE, node->GetName( ) );
548
549 Utils::CopyNode( node, newTargetNode );
550 wxXmlNode* root = m_mergeList->GetTargetRoot( );
551 Catalog::AddEntry( root, newTargetNode, 0 );
552 }
553 else
554 {
555 for ( int i = 0; i < Catalog::DT_NbrTypes; i++ )
556 {
557 m_compEntry[ i ]->CopyAttribute( );
558 }
559 }
560 event.Skip( );
