196wxTreeItemId CompareTreeCtrl::AddChild( wxTreeItemId parent, wxXmlNode* child )
197{
198 Catalog::IconID icon;
199 wxString label;
200 Catalog::CatalogBaseType nodeType;
201 CompareTreeItemData* itemData = CreateChildData( child, 0, label, icon, nodeType );
202
203 if ( !itemData )
204 {
205 return 0;
206 }
207 wxTreeItemId childID = AppendItem( parent, label, icon, -1, itemData );
208
209
210
211 if ( nodeType == Catalog::NT_Entry )
212 {
213 Catalog::MergeData* entry = new Catalog::MergeData( m_comparePanel );
214 entry->SetTargetEntry( child );
215 Catalog::MergeStatus status = entry->SetStatus( );
216
217 Catalog::Entry* catEntry = new Catalog::Entry( child );
218 Catalog::IconID icon = GetIconId( catEntry, status );
219 //set the icon for the appropriate state
220 SetItemImage( childID, icon );
221
222 }
223 else
224 {
225 // not a entry just add the appropriate image
226 SetItemImage( childID, Catalog::Icon_Folder );
227 }
228
229
230 // now do it all again for this entrys children
231 wxXmlNode* grandChild = child->GetChildren( );
232 while ( grandChild )
233 {
234
235 // start adding child elements to it.
236 AddChild( childID, grandChild );
237 grandChild = grandChild->GetNext( );
238 }
239 SetTreeItemCollapseState( childID );
240 return childID;
241}

272CompareTreeItemData* CompareTreeCtrl::CreateChildData( wxXmlNode* targetNode, wxXmlNode* mergeNode,
273 wxString& label,
274 Catalog::IconID& icon,
275 Catalog::CatalogBaseType& nodeType )
276{
277 wxXmlNode* node = targetNode;
278 if ( !targetNode )
279 {
280 node = mergeNode;
281 }
282
283 wxString name = node->GetName( );
284 nodeType = Catalog::FindCatalogBaseType( name );
285 //wxString label;
286 //Catalog::IconID icon;
287
288 //if the child element is not a entry
289 //if ( !name.Cmp( Catalog::CatalogBaseNames[ Catalog::NT_Entry ] ) )
290 if ( nodeType == Catalog::NT_Entry )
291 {
292 // then we add the appropriate icon and label
293 Catalog::Entry entry( node );
294 // entry combines the entryID and its name to form a label
295 label = entry.GetLabel( );
296

298 }

300 {
