193void DefinePeriodsPanel::Update( )
194{
195 if ( m_antiqueTextBox->IsModified( ) )
196 {
197 GetSettings( )->SetDirty( );
198 GetSettings( )->SetLowerPeriod( m_antiqueTextBox->GetValue( ) );
199 m_antiqueTextBox->SetModified( false );
200 }
201 if ( m_AntiqueYear->IsModified( ) )
202 {
203 GetSettings( )->SetDirty( );
204 GetSettings( )->SetLowerDivision( m_AntiqueYear->GetValue( ) );
205 m_AntiqueYear->SetModified( false );
206 }
207 if ( m_modernTextBox->IsModified( ) )
208 {
209 GetSettings( )->SetDirty( );
210 GetSettings( )->SetUpperPeriod( m_modernTextBox->GetValue( ) );
211 m_modernTextBox->SetModified( false );
212 }
213 if ( m_classicTextBox->IsModified( ) )
214 {
215 GetSettings( )->SetDirty( );
216 GetSettings( )->SetMiddlePeriod( m_classicTextBox->GetValue( ) );
217 m_classicTextBox->SetModified( false );

221 GetSettings( )->SetDirty( );
