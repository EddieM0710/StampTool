289 LayoutBase* GetSelectedNodePage( )
290 {
291 AlbumVolume* albumVolume = Design::GetAlbumVolume( );
292 if ( albumVolume )
293 {
294 AlbumTreeCtrl* treeCtrl = GetAlbumTreeCtrl( );
295 if ( treeCtrl )
296 {
297 wxTreeItemId id = treeCtrl->GetSelection( );
298 if ( id.IsOk( ) )
299 {
300 DesignTreeItemData* data = ( DesignTreeItemData* ) treeCtrl->GetItemData( id );
301 while ( id.IsOk( ) && data )
302 {
303 Design::AlbumBaseType type = data->GetType( );
304 if ( type == Design::AT_Album )
305 {
306 wxTreeItemIdValue cookie;
307 wxTreeItemId childID = treeCtrl->GetFirstChild( id, cookie );
308 if ( childID.IsOk( ) )
309 {
310 Design::AlbumBaseType childType = treeCtrl->GetItemType( childID );
311 data = ( DesignTreeItemData* ) treeCtrl->GetItemData( childID );
312 return ( LayoutBase* ) data->GetNodeElement( );
313 }
314 }
315 else if ( type == AT_Page )
316 {
317 return ( LayoutBase* ) data->GetNodeElement( );
318 }
319 id = treeCtrl->GetItemParent( id );
320 if ( id.IsOk( ) )
321 {
322 data = ( DesignTreeItemData* ) treeCtrl->GetItemData( id );
323 }
324 }
325 }
326 }
327 }
328 return ( LayoutBase* ) 0;
329 }

359 AlbumBase* MakeNode( wxXmlNode* node )
360 {
361 wxString nodeName = node->GetName( );
362 AlbumBaseType type = FindAlbumBaseType( nodeName );
363
364 AlbumBase* object = ( AlbumBase* ) 0;
365
366 if ( type == AT_Album )
367 {
368 object = ( AlbumBase* )new Album( node );
369 }
370 else if ( type == AT_Page )
371 {
372 object = ( AlbumBase* )new Page( node );
373 }
374 else if ( type == AT_Col )
375 {
376 object = ( AlbumBase* )new Column( node );
377 }
378 else if ( type == AT_Row )
379 {
380 object = ( AlbumBase* )new Row( node );
381 }
382 else if ( type == AT_Stamp )
383 {
384 object = ( AlbumBase* )new Stamp( node );
385 }
386 return object;
