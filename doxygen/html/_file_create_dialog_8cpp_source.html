152void FileCreateDialog::OnOKClick( wxCommandEvent& event )
153{
154 m_path = m_fileCtrl->GetPath( );
155 m_file = m_fileCtrl->GetFilename( );
156 m_dir = m_fileCtrl->GetDirectory( );
157 wxFileName sptFile( m_path );
158 if ( sptFile.Exists( ) )
159 {
160 wxMessageDialog* dlg = new wxMessageDialog(
161 this,
162 wxT( "The selected file exists. \n"\
163 "Select \"OK\" to overwrite the existing file losing the contents.\n"\
164 "Or select \"Cancel\" to choose another name.\n" ),
165 wxT( " Warning! File Exists.\n" ),
166 wxOK | wxCANCEL | wxCENTER );
167 int rsp = dlg->ShowModal( );
168 if ( rsp == wxID_CANCEL )
169 {
170 return;
171 }
172 }
173
174 event.Skip( );
175}
