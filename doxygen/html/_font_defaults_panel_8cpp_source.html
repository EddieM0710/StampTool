110void FontDefaultsPanel::CreateControls( )
111{
112 FontDefaultsPanel* thePanel = this;
113 int lastID = ID_LastID;
114
115 wxBoxSizer* detailsHorizontalSizer = new wxBoxSizer( wxHORIZONTAL );
116 thePanel->SetSizer( detailsHorizontalSizer );
117
118 wxBoxSizer* leftColumnVerticalSizer = new wxBoxSizer( wxVERTICAL );
119 detailsHorizontalSizer->Add( leftColumnVerticalSizer, 1, wxGROW | wxALL, 0 );
120
121 wxBoxSizer* middleColumnVerticalSizer = new wxBoxSizer( wxVERTICAL );
122 detailsHorizontalSizer->Add( middleColumnVerticalSizer, 1, wxGROW | wxALL, 0 );
123
124 wxBoxSizer* rightColumnVerticalSizer = new wxBoxSizer( wxVERTICAL );
125 detailsHorizontalSizer->Add( rightColumnVerticalSizer, 1, wxGROW | wxALL, 0 );
126
127 SetupFontPicker( thePanel, leftColumnVerticalSizer, lastID,
128 _( "Title Font" ), _( "Default" ),
129 m_titleFontPicker, m_titleColorPicker,
130 wxFontPickerEventHandler( FontDefaultsPanel::OnTitleFontPicker ),
131 wxColourPickerEventHandler( FontDefaultsPanel::OnTitleColorPicker ),
132 wxCommandEventHandler( FontDefaultsPanel::OnTitleDefaultClick ) );
133
134 SetupFontPicker( thePanel, leftColumnVerticalSizer, lastID,
135 _( "SubTitle Font" ), _( "Default" ),
136 m_subTitleFontPicker, m_subTitleColorPicker,
137 wxFontPickerEventHandler( FontDefaultsPanel::OnSubTitleFontPicker ),
138 wxColourPickerEventHandler( FontDefaultsPanel::OnSubTitleColorPicker ),
139 wxCommandEventHandler( FontDefaultsPanel::OnSubTitleDefaultClick ) );
140
141 SetupFontPicker( thePanel, middleColumnVerticalSizer, lastID,
142 _( "Text Font" ), _( "Default" ),
143 m_textFontPicker, m_textColorPicker,
144 wxFontPickerEventHandler( FontDefaultsPanel::OnTextFontPicker ),
145 wxColourPickerEventHandler( FontDefaultsPanel::OnTextColorPicker ),
146 wxCommandEventHandler( FontDefaultsPanel::OnTextDefaultClick ) );
147
148 SetupFontPicker( thePanel, middleColumnVerticalSizer, lastID,
149 _( "Default Catalog Nbr Font" ), _( "Default" ),
150 m_nbrFontPicker, m_nbrColorPicker,
151 wxFontPickerEventHandler( FontDefaultsPanel::OnNbrFontPicker ),
152 wxColourPickerEventHandler( FontDefaultsPanel::OnNbrColorPicker ),
153 wxCommandEventHandler( FontDefaultsPanel::OnNbrDefaultClick ) );
154
155 SetupFontPicker( thePanel, rightColumnVerticalSizer, lastID,
156 _( "Default Name Font" ), _( "Default" ),
157 m_nameFontPicker, m_nameColorPicker,
158 wxFontPickerEventHandler( FontDefaultsPanel::OnNameFontPicker ),

232void FontDefaultsPanel::OnTitleDefaultClick( wxCommandEvent& event )
233{
234 int ndx = Design::GetAlbum( )->GetFontNdx( Design::AT_TitleFontType );
235 Utils::FontList* fontList = GetFontList( );
236 wxFont font = fontList->GetFont( ndx );
237 wxColour color = fontList->GetColor( ndx );
238 m_titleFontPicker->SetSelectedFont( font );
239 m_titleColorPicker->SetColour( color );
240
241 Design::GetAlbum( )->SetFont( Design::AT_TitleFontType, font, color );
242
243 event.Skip( );
244}
