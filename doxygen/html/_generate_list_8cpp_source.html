171bool GenerateList::CheckStatus( Catalog::Entry* entry )
172{
173 if ( m_checkedStatusItems == 0 )
174 {
175 return true;
176 }
177 wxString emission = entry->GetInventoryStatus( );
178 int i = Catalog::FindStatusType( emission );
179 if ( i >= 0 )
180 {
181 int base = 1;
182 base = base <<= i;
183 return m_checkedStatusItems & base;
184 }
185 return true;
186}

298int MakeBitPatern( wxArrayInt& array )
299{
300 int nbr = array.Count( );
301 int val = 0;
302 for ( int i = 0; i < nbr; i++ )
303 {
304 int base = 1;
305 base = base <<= array[ i ];
306 val = val | base;
307 }
308 return val;
309}

440void GenerateList::ShowRow( Catalog::Entry* entry, int row )
441{
442 wxFont font = m_gridCtrl->GetCellFont( row, 2 );
443
444 m_gridCtrl->AppendRows( 1 );
445 m_gridCtrl->SetCellValue( row, 0, entry->GetID( ) );
446 m_gridCtrl->SetCellValue( row, 1, entry->GetInventoryStatus( ) );
447 m_gridCtrl->SetCellValue( row, 2, entry->GetName( ) );
448
449 int size = GetStringLen( font, entry->GetName( ) );
450 int curr = m_gridCtrl->GetColSize( 2 );
451 if ( size > curr )
452 {
453 m_gridCtrl->SetColSize( 2, size );
454 }
455
456 m_gridCtrl->SetCellValue( row, 3, entry->GetIssuedDate( ) );
457 m_gridCtrl->SetCellValue( row, 4, entry->GetEmission( ) );
458 size = GetStringLen( font, entry->GetEmission( ) );
459 curr = m_gridCtrl->GetColSize( 4 );
460 if ( size > curr )
461 {
462 m_gridCtrl->SetColSize( 4, size );
463 }
464 m_gridCtrl->SetCellValue( row, 5, entry->GetFormat( ) );
465 m_gridCtrl->SetCellValue( row, 6, entry->GetWidth( ) );
466 m_gridCtrl->SetCellValue( row, 7, entry->GetHeight( ) );
467 m_gridCtrl->SetCellValue( row, 8, entry->GetMount( ) );
468}
