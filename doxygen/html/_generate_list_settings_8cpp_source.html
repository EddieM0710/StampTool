241int GenerateListSettings::MakeBitPatern( wxArrayInt& array )
242{
243 int nbr = array.Count( );
244 int val = 0;
245 for ( int i = 0; i < nbr; i++ )
246 {
247 int base = 1;
248 base = base <<= array[ i ];
249 val = val | base;
250 }
251 return val;
252}
