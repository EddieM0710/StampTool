208HelperPanel::TitleHelper* HelperPanel::SetupTitleHelper( wxWindow* parent, wxBoxSizer* inSizer, int& idLast,
209 int style,
210 wxObjectEventFunction titleCheckBoxEventHandler,
211 wxObjectEventFunction titleEventHandler,
212 wxObjectEventFunction subTitleCheckBoxEeventHandler,
213 wxObjectEventFunction subTitleEventHandler )
214{
215 TitleHelper* titleHelper = new TitleHelper;
216
217 wxBoxSizer* horizontalSizer = new wxBoxSizer( wxHORIZONTAL );
218 inSizer->Add( horizontalSizer, 0, wxGROW | wxALL, 5 );
219 titleHelper->titleCheckbox = 0;
220 titleHelper->subTitleCheckbox = 0;
221 titleHelper->titleLabel = 0;
222 titleHelper->subTitleLabel = 0;
223
224 titleHelper->titleCheckbox = SetupCheckBox( parent, horizontalSizer, idLast, _( "Show Title" ), titleCheckBoxEventHandler );
225 titleHelper->titleCheckbox->SetValue( false );
226 if ( style & HasSubTitle )
227 {
228 titleHelper->subTitleCheckbox = SetupCheckBox( parent, horizontalSizer, idLast, _( "Show SubTitle" ), subTitleCheckBoxEeventHandler );
229 titleHelper->subTitleCheckbox->SetValue( false );
230 }
231 if ( style & HasTextCtrl )
232 {
233 titleHelper->titleLabel = SetupLabelText( parent, inSizer, idLast, _( "Title" ), true, titleEventHandler );
234 if ( style & HasSubTitle )
235 {
236 titleHelper->subTitleLabel = SetupLabelText( parent, inSizer, idLast, _( "Sub Title" ), true, subTitleEventHandler );
237 }
238 }
239 return titleHelper;
240}

261void HelperPanel::UpdateTitleState( HelperPanel::TitleHelper* titleHelper )
262{
263 bool state = titleHelper->titleCheckbox->GetValue( );
264 titleHelper->titleCheckbox->Show( true );
265 if ( state )
266 {
267 if ( titleHelper->titleLabel )
268 {
269 titleHelper->titleLabel->Enable( state );
270 }
271 if ( titleHelper->subTitleCheckbox )
272 {
273 titleHelper->subTitleCheckbox->Enable( );
274 }
275 }
276 else
277 {
278 if ( titleHelper->titleLabel )
279 {
280 titleHelper->titleLabel->Disable( );
281 }
282 if ( titleHelper->subTitleCheckbox )
283 {
284 titleHelper->subTitleCheckbox->Disable( );
285 titleHelper->subTitleCheckbox->SetValue( state );
286 if ( titleHelper->subTitleLabel )
287 {
288 titleHelper->subTitleLabel->Disable( );
289 }
290 }
291 }
292 if ( titleHelper->subTitleLabel )
293 {
294 titleHelper->subTitleLabel->Show( );
295 }
296}
