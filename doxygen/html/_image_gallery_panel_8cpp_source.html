160void ImageGalleryPanel::SetBitmap( wxDC& dc, wxString filename, int& row, int& col )
161{
162 wxImage image;
163 if ( filename.IsEmpty( ) )
164 {
165 return;
166 //image = wxImage( NotFound );
167 }
168 else
169 {
170 //wxFileName fn( filename );
171 wxString str = GetProject( )->GetImageFullPath( filename );
172
173 if ( !GetProject( )->ImageExists( str ) )
174 {
175 return;
176 //image = wxImage( NotFound );
177 }
178 else
179 {
180 image = wxImage( str );
181
182 if ( !image.IsOk( ) )
183 {
184 return;
185 //image = wxImage( NotFound );
186 }
187 }
188 }
189
190 // m_bitmap = image;
191 m_zoom = .9;
192
193 double cellWidth = 1.5 * 25.4;
194 double cellHeight = 1.5 * 25.4;
195
196 double scale = 1;
197 if ( image.GetWidth( ) > image.GetHeight( ) )
198 {
199 scale = cellWidth / image.GetWidth( );
200 }
201 else
202 {
203 scale = cellHeight / image.GetHeight( );
204 }
205
206 DrawImage( dc, image,
207 col * cellWidth * m_zoom, row * cellHeight * m_zoom,
208 image.GetWidth( ) * scale * m_zoom, image.GetHeight( ) * scale * m_zoom );
209
210 col++;
211 if ( col > 7 )
212 {
213 col = 0;
214 row++;
215 }
