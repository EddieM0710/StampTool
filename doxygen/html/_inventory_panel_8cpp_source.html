246{

319void InventoryPanel::OnContextPopup( wxCommandEvent& event )
320{
321 int id = event.GetId( );
322 switch ( id )
323 {
324 case ID_AddItem:
325 {
326 AddRow( AddNewInventoryItem( ) );
327 break;
328 }
329 case ID_DeleteItem:
330 {
331 break;
332 }
333 case ID_MoveItemUp:
334 {
335 wxArrayInt array = m_grid->GetSelectedRows( );
336 m_grid->IsSelection( );
337 int row = array[ 0 ];
338 if ( row > 0 && row < m_specimenList.size( ) )
339 {
340 int prevRow = row - 1;
341 wxXmlNode* node = m_specimenList[ row ];
342 wxXmlNode* prevNode = m_specimenList[ prevRow ];
343 wxXmlNode* parent = node->GetParent( );
344 parent->RemoveChild( node );
345 parent->InsertChild( node, prevNode );
346 UpdateInventoryGrid( );
347 }
348 break;
349 }
350 case ID_MoveItemBottom:
351 {
352 wxArrayInt array = m_grid->GetSelectedRows( );
353 m_grid->IsSelection( );
354 int row = array[ 0 ];
355 if ( row >= 0 && row < m_specimenList.size( ) - 1 )
356 {
357 int lastRow = m_specimenList.size( ) - 1;
358 wxXmlNode* node = m_specimenList[ row ];
359 wxXmlNode* lastNode = m_specimenList[ lastRow ];
360 wxXmlNode* parent = node->GetParent( );
361 parent->RemoveChild( node );
362 parent->InsertChildAfter( node, lastNode );
363 UpdateInventoryGrid( );
364 }
365 break;
366 }
367 case ID_MoveItemDown:
368 {
369 wxArrayInt array = m_grid->GetSelectedRows( );
370 m_grid->IsSelection( );
371 int row = array[ 0 ];
372 if ( row >= 0 && row < m_specimenList.size( ) - 1 )
373 {
374 int nextRow = row + 1;
375 wxXmlNode* node = m_specimenList[ row ];
376 wxXmlNode* nextNode = m_specimenList[ nextRow ];
377 wxXmlNode* parent = node->GetParent( );
378 parent->RemoveChild( node );
379 parent->InsertChildAfter( node, nextNode );
380 UpdateInventoryGrid( );
381 }
382 break;
383 }
384 case ID_MoveItemTop:
385 {
386 wxArrayInt array = m_grid->GetSelectedRows( );
387 m_grid->IsSelection( );
388 int row = array[ 0 ];
389 if ( row > 0 && row < m_specimenList.size( ) )
390 {
391 int firstRow = 0;
392 wxXmlNode* node = m_specimenList[ row ];
393 wxXmlNode* firstNode = m_specimenList[ firstRow ];
