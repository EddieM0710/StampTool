170 void LayoutBase::ValidateChildType( int& nbrRows, int& nbrCols, int& nbrLeaf )
171 {
172 // count the number of rows/cols planned
173 nbrRows = 0;
174 nbrCols = 0;
175 nbrLeaf = 0;
176
177 wxTreeItemIdValue cookie;
178 wxTreeItemId parentID = GetTreeItemId( );
179 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
180 while ( childID.IsOk( ) )
181 {
182 AlbumBaseType type = ( AlbumBaseType ) GetAlbumTreeCtrl( )->GetItemType( childID );
183 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
184 switch ( type )
185 {
186 case AT_Row:
187 {
188 nbrRows++;
189 break;
190 }
191 case AT_Col:
192 {
193 nbrCols++;
194 break;
195 }
196 case AT_Stamp:
197 {
198 nbrLeaf++;
199 break;
200 }
201 case AT_Text:
202 {
203 nbrLeaf++;
204 break;
205 }
206 }
207 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
208 }
209 if ( ( nbrRows > 0 ) && ( nbrCols > 0 ) )
210 {
211 ReportLayoutError( "ValidateChildType", "Only Rows or Columns are allowed as the children, not both" );
212 }
213 }
