119 MergeStatus MergeData::SetStatus( ){
120 if ( !m_targetEntry || m_targetEntry == ( Entry* ) 0x8000000000000000 )
121 {
122 m_status = MS_TargetMissing;
123 return m_status;
124 }
125 else if ( !m_mergeEntry || m_mergeEntry == ( Entry* ) 0x8000000000000000 )
126 {
127 m_status = MS_MergeMissing;

129 }
130 for ( int i = 0; i < DT_NbrTypes; i++ )
131 {
132 if ( cmp( ( DataTypes ) i ) )
133 {
134 m_status = MS_Different;
135 return m_status;
136 }
137 else
138 {
139 m_status = MS_Same;
140 }
141 }
142 return MS_Undefined;

322 MergeData* MergeList::FindTargetEntryForNewEntry( Entry* entry, DataTypes type )
323 {
324 wxString entryVal = entry->GetAttr( type );
325 for ( MergeEntryList::iterator it = std::begin( m_mergeList );
326 it != std::end( m_mergeList );
327 ++it )
328 {
329 MergeData* mergeEntry = ( MergeData* ) ( *it );
330 if ( !entryVal.Cmp( mergeEntry->GetTargetAttribute( type ) ) )
331 {
332 return mergeEntry;
333 }
334 }
335 return ( MergeData* ) 0;
