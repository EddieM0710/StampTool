180 bool MountCSVData::FixUpLine( wxString& line, int lineNbr )
181 {
182 int curr = 0;
183 int last = line.length( );
184 int firstQuote = 0;
185 int nextQuote = 0;
186 int comma = 0;
187 while ( curr < last )
188 {
189 firstQuote = line.find( "\"", curr );
190 if ( firstQuote < 0 )
191 {
192 return true;
193 }
194 nextQuote = line.find_first_of( "\"", firstQuote + 1 );
195 if ( nextQuote < 0 )
196 {
197 // unmatched parens
198 // this means the csv is messed up
199
200 std::cout << "Format error in csv file " << m_filename << "at line number " << lineNbr << " " << line << "\n";
201
202 return false;
203 }
204 comma = line.find_first_of( ",", firstQuote );
205 bool check_again = true;
206 while ( check_again )
207 {
208 if ( ( comma > firstQuote ) && ( comma < nextQuote ) )
209 {
210 line = line.replace( comma, 1, "{" );
211 comma = line.find_first_of( ",", firstQuote );
212 }
213 else
214 {
215 check_again = false;
216 }
217 }
218 curr = nextQuote + 1;
219 }
220 return true;
221 };
