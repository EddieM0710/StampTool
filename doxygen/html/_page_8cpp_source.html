403 bool Page::UpdateMinimumSize( )
404 {
405
406 int nbrRows = 0;
407 int nbrCols = 0;
408 int nbrStamps = 0;
409 m_pageType = Design::PT_None;
410 SetXPos( GetLeftPageMargin( ) );
411 SetYPos( GetTopPageMargin( ) );
412 if ( Design::IsPortrait( Orientation( ) ) )
413 {
414 SetWidth( GetPageWidth( ) - GetRightPageMargin( ) - GetLeftPageMargin( ) );
415 SetHeight( GetPageHeight( ) - GetTopPageMargin( ) - GetBottomPageMargin( ) );
416 }
417 else
418 {
419 SetWidth( GetPageHeight( ) - GetRightPageMargin( ) - GetLeftPageMargin( ) );
420 SetHeight( GetPageWidth( ) - GetTopPageMargin( ) - GetBottomPageMargin( ) );
421 }
422 double minWidth = 0.0;
423 double minHeight = 0.0;
424 // the page LayoutBase m_frame contains the parameters for the page.
425 // the actual content of the page is the area inside the border, i.e., the ContentFrame
426 SetContentFrame( );
427
428 wxTreeItemIdValue cookie;
429 wxTreeItemId parentID = GetTreeItemId( );
430 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
431 while ( childID.IsOk( ) )
432 {
433 AlbumBaseType type = ( AlbumBaseType ) GetAlbumTreeCtrl( )->GetItemType( childID );
434
435 // set the page type based on the first row or col child
436 if ( m_pageType == Design::PT_None )
437 {
438 if ( type == Design::AT_Row )
439 {
440 // Rows have cols and cols have rows
441 // If we found a row this page is behaving like a column
442 // There should only be rows as first level children of this page
443 m_pageType = Design::PT_ColumnPage;
444 }
445 else if ( type == Design::AT_Col )
446 {
447 m_pageType = Design::PT_RowPage;
448 }
449 }
450
451 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
452
453 // calc the min size for this items children
454 child->UpdateMinimumSize( );
455
456 if ( m_pageType == Design::PT_RowPage )
457 {
458 //positioning across the page
459 //the min height of the page is the size of the tallest child
460 //the min width is the sum of the min widths of the children
461
462 // keep the max min height
463 if ( child->GetMinHeight( ) > minHeight )
464 {
465 minHeight = child->GetMinHeight( );
466 }
467 // keep running total of min child width
468 minWidth += child->GetMinWidth( );
469 }
470 else // PT_ColumnPage
471 {
472 // positioning down the page
473 //the min width of the page is the size of the widest child
474 //the min height is the sum of the min heights of the children
475 if ( child->GetMinWidth( ) > minWidth )
476 {
477 minWidth = child->GetMinWidth( );
478 }
479 minHeight += child->GetMinHeight( );
480 }
481 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
482 }
483
484 //if this item shows its title then add that to the min height
485 GetTitleFrame( )->Init( );
486 if ( GetShowTitle( ) )
487 {
488 m_titleFrame->UpdateString( minWidth, minWidth );
489 // Add the title height
490 minHeight += GetTitleFrame( )->GetHeight( );
491
492 GetTitleFrame( )->SetYPos( 0 );
493 }
494
495 SetMinWidth( minWidth
496 + GetLeftContentMargin( )
497 + GetRightContentMargin( ) );
498 SetMinHeight( minHeight
499 + GetTopContentMargin( )
500 + GetBottomContentMargin( ) );
501
502 GetErrorArray( )->Empty( );
503
504 return ( ValidateNode( ) != AT_FATAL );
505 }

509 void Page::UpdatePositions( )
510 {
511 // calc space used by children
512 // this means looping through the children and getting the actual sizes
513 // of the children, not their calculated min size
514 wxTreeItemIdValue cookie;
515 wxTreeItemId parentID = GetTreeItemId( );
516 LayoutBase* parent = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( parentID );
517 int nbrChildren = parent->GetNbrChildren( );
518
519 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
520 double actualWidth = 0;
521 double actualHeight = 0;
522 while ( childID.IsOk( ) )
523 {
524 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
525 if ( m_pageType == Design::PT_ColumnPage )
526 {
527 actualHeight += child->GetHeight( );
528 }
529 else
530 {
531 actualWidth += child->GetWidth( );
532 }
533 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
534 }
535
536 double xPos = 0;
537 double yPos = 0;
538 //First calc title position
539 if ( GetShowTitle( ) )
540 {
541 double titleXPos = ( m_contentFrame.GetWidth( ) - GetTitleFrame( )->GetWidth( ) ) / 2;
542
543 if ( titleXPos < 0 )
544 {
545 titleXPos = 0;
546 }
547 GetTitleFrame( )->SetXPos( titleXPos );
548
549 GetTitleFrame( )->SetYPos( GetTopContentMargin( ) );
550
551 // allow for space above title, title height and that much again for nice spaing
552 yPos = GetTitleFrame( )->GetHeight( ) + 2 + GetTopContentMargin( );
553 }
554 else
555 {
556 GetTitleFrame( )->Init( );
557 yPos = GetTopContentMargin( );
558 }
559
560 // if there are rows then we are positioning them down the page
561 double spacing = 0;
562 double totalExtraSpace = 0;
563 if ( m_pageType == Design::PT_ColumnPage )
564 {
565 totalExtraSpace = m_contentFrame.GetHeight( )
566 - actualHeight
567 - GetTopContentMargin( )
568 - GetBottomContentMargin( );
569 if ( GetShowTitle( ) )
570 {
571 totalExtraSpace -= GetTitleFrame( )->GetHeight( );
572 }
573
574 spacing = totalExtraSpace / ( nbrChildren + 1 );
575 }
576 else // we are positioning them across the page
577 {
578 totalExtraSpace = m_contentFrame.GetWidth( ) - actualWidth;
579 spacing = totalExtraSpace / ( nbrChildren + 1 );
580
581 }
582
583 //figure out starting pos accounting for title if present
584 if ( m_pageType == Design::PT_ColumnPage )
585 {
586 yPos += spacing;
587 }
588 else //PT_RowPage || Stamp
589 {
590 xPos += spacing;
591 }
592
593 childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
594 while ( childID.IsOk( ) )
595 {
596 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
597 child->UpdatePositions( );
598
599 if ( m_pageType == Design::PT_RowPage )
600 {
601 child->SetXPos( xPos );
602 child->SetYPos( 0 );
603 xPos = xPos + spacing + child->GetWidth( );
604 }
605 else // PT_ColumnPage
606 {
607 child->SetXPos( 0 );
608 child->SetYPos( yPos );
609 yPos = yPos + spacing + child->GetHeight( );
610 }
611
612 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
613 }
614 }
