520void PreferencesDialog::UpdateDetails()
521{
522
523 if ( m_imageDirectory->IsModified( ) )
524 {
525 GetSettings( )->SetDirty( );
526 GetProject( )->SetImageDirectory( m_imageDirectory->GetValue( ) );
527 m_imageDirectory->SetModified( false );
528 }
529
530 if ( m_country->IsModified( ) )
531 {
532 GetSettings( )->SetDirty( );
533 GetSettings( )->SetCountryID( m_country->GetValue( ) );
534 m_country->SetModified( false );
535 }
536
537
538 if ( m_catalog->IsModified( ) )
539 {
540 GetSettings( )->SetDirty( );
541 GetSettings( )->SetCatalogID( m_catalog->GetValue( ) );
542 m_catalog->SetModified( false );
543 }
544
545 if ( m_loadLastFileAtStartUp->GetValue( ) != GetSettings( )->GetLoadLastFileAtStartUp( ) )
546 {
547 GetSettings( )->SetDirty( );
548 GetSettings( )->SetLoadLastFileAtStartUp( m_loadLastFileAtStartUp->GetValue( ) );
549 }
550
551
552 if ( m_recentListSize->IsModified( ) )
553 {
554 GetSettings( )->SetDirty( );
555 long val;
556 if ( m_recentListSize->GetValue( ).ToLong( &val ) )
557 {
558 GetSettings( )->SetNbrRecentPreference( val );
559 }
560 else
561 {
562 GetSettings( )->SetNbrRecentPreference( 1 );
563 }
564 m_recentListSize->SetModified( false );
565 }
