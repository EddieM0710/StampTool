385 void Project::LoadAttributes( wxXmlNode* thisObject )
386 {
387 const wxXmlAttribute* attr = thisObject->GetAttributes( );
388 wxFileName filename;
389 while ( attr )
390 {
391 wxString name = attr->GetName( );
392 wxString val = attr->GetValue( );
393

466 void Project::Save( )
467 {
468
469 if ( m_projectFilename.IsEmpty( ) )
470 {
471 //query for valid filename
472 m_projectFilename = "UndefinedName.spt";
473 }
474 if ( wxFileExists( m_projectFilename ) )
475 {
476 wxFileName bakFile( m_projectFilename );
477 bakFile.SetExt( "bak.spt" );
478 wxRenameFile( m_projectFilename, bakFile.GetFullName( ), true );
479 }
480 if ( m_ProjectDoc )
481 {
482 delete m_ProjectDoc;
483 m_ProjectDoc = 0;
484 }
485 wxXmlDocument* newDoc = MakeDefaultProjectDocument( );
486
487 wxXmlNode* root = newDoc->GetRoot( );
488
489 // set curr vals
490 SetAttrStr( root, "Country", GetProjectCountryID( ) );
491 SetAttrStr( root, "CatalogCode", GetProjectCatalogCode( ) );
492 SetAttrStr( root, "ImagePath", m_imageDirectory );
493
494 wxXmlNode* albumListNode = FirstChildElement( root, "AlbumList" );
495
496 // add the entries of the allbum TOC
497 GetAlbumTOCTreeCtrl( )->MakeTree( albumListNode );
498
499 wxXmlNode* catalogListNode = FirstChildElement( root, "CatalogList" );
500
501 // add the entries of the catalog TOC
502 GetCatalogTOCTreeCtrl( )->MakeTree( catalogListNode );
503
504 if ( m_ProjectDoc )
505 {
506 delete m_ProjectDoc;
507 }
508 m_ProjectDoc = newDoc;
509
510 // save the new project file
511 m_ProjectDoc->Save( m_projectFilename );
512
513 // then save all the catalogs and albums
514 GetCatalogData( )->FileSave( );
515 GetAlbumData( )->FileSave( );
516
517 SetDirty( false );
518 }
