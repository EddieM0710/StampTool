305 bool Row::UpdateMinimumSize( )
306 {
307 //Positioning across the row.
308 //The min height of the row is the size of the tallest child
309 //plus the title if used
310 //The min width is the sum of the min widths of the children.
311 double minWidth = 0.0;
312 double minHeight = 0.0;
313
314 wxTreeItemIdValue cookie;
315 wxTreeItemId parentID = GetTreeItemId( );
316 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
317 while ( childID.IsOk( ) )
318 {
319
320 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
321
322 // calc the min size ror this items children
323 child->UpdateMinimumSize( );
324
325
326 // keep the max min height
327 if ( child->GetMinHeight( ) > minHeight )
328 {
329 minHeight = child->GetMinHeight( );
330 }
331
332 // keep running total of min child width
333 minWidth += child->GetMinWidth( );
334 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
335 }
336
337 //if this item shows its title then add that to the min height
338 GetTitleFrame( )->Init( );
339 if ( GetShowTitle( ) )
340 {
341 // update the title frame
342 m_titleFrame->UpdateString( minWidth, minWidth );
343
344 // Add the title height
345 minHeight += GetTitleFrame( )->GetHeight( );
346
347 GetTitleFrame( )->SetYPos( 0 );
348 }
349
350 // row min height and width
351 SetMinHeight( minHeight
352 + GetTopContentMargin( )
353 + GetBottomContentMargin( ) );
354
355 SetMinWidth( minWidth
356 + GetLeftContentMargin( )
357 + GetRightContentMargin( ) );
358
359 GetErrorArray( )->Empty( );
360
361 return true;
362 }

366 // calculate the row layout based on child parameters
367 void Row::UpdatePositions( )
368 {
369 // go to the bottom of each child container object ( row, column, page )
370 // and begin filling in position relative to the parent
371
372 // calc space used by children
373 // this means looping through the children and getting the actual sizes
374 // of the children, not their calculated min size
375 double actualWidth = 0;
376 double maxHeight = 0;
377 double maxTitleHeight = 0;
378 double maxFrameHeight = 0;
379
380 wxTreeItemIdValue cookie;
381 wxTreeItemId parentID = GetTreeItemId( );
382 LayoutBase* parent = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( parentID );
383 int nbrChildren = parent->GetNbrChildren( );
384
385 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
386
387 while ( childID.IsOk( ) )
388 {
389 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
390 actualWidth += child->GetWidth( );
391 if ( child->GetHeight( ) > maxHeight )
392 {
393 maxHeight = child->GetHeight( );
394 }
395 if ( child->IsNodeType( AT_Stamp ) )
396 {
397 Stamp* stamp = ( Stamp* ) child;
398 //std::cout << stamp->GetNameFrame( )->GetString( ) << " max:" << maxTitleHeight << " found:" << stamp->GetNameFrame( )->GetHeight( );
399
400 if ( stamp->GetNameFrame( )->GetHeight( ) > maxTitleHeight )
401 {
402 maxTitleHeight = stamp->GetNameFrame( )->GetHeight( );
403 }
404 //std::cout << " new max:" << maxTitleHeight << "\n";
405
406 if ( stamp->GetBorderFrame( )->GetHeight( ) > maxFrameHeight )
407 {
408 maxFrameHeight = stamp->GetBorderFrame( )->GetHeight( );
409 }
410 }
411 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
412 }
413
414 double xPos = 0;
415 double yPos = 0;
416
417 //First calc title position
418 if ( GetShowTitle( ) )
419 {
420 GetTitleFrame( )->SetXPos( ( GetWidth( ) - GetTitleFrame( )->GetWidth( ) - GetLeftContentMargin( ) ) / 2 );
421 GetTitleFrame( )->SetYPos( GetRightContentMargin( ) );
422 // Children start just below the title Frame
423 // allow for space above title and space above and below it
424 yPos = GetTitleFrame( )->GetHeight( );// + GetTopContentMargin( )+ GetBottomContentMargin( );
425 }
426 else
427 {
428 GetTitleFrame( )->Init( );
429 yPos = GetTopContentMargin( );
430 }
431
432 double spacing = 4;
433 // this is a row so we are positioning children across the page
434 if ( CalculateSpacing( ) )
435 {
436 spacing = ( GetWidth( )
437 - GetLeftContentMargin( )
438 - GetRightContentMargin( )
439 - actualWidth )
440 / ( nbrChildren + 1 );
441
442 // inital x/y pos within the row
443 xPos = spacing + GetLeftContentMargin( );
444
445 }
446 else
447 {
448 spacing = GetFixedSpacingDbl( );
449 //find the extra space to go on each end
450 //it is the amount from the width of items - the minwidth - amount of total space between items
451 //then divide the remaining by 2 to go on each end
452 xPos = (
453 GetWidth( ) // width of object
454 - GetLeftContentMargin( ) //less left content
455 - GetRightContentMargin( ) //less right content
456 - actualWidth // actual width used by children
457 - ( nbrChildren - 1 ) * spacing //less the width between the children
458 )
459 / 2; // and leave half on the beginning of the row
460
461 }
462
463 if ( xPos < 0 )xPos = 0;
464 if ( yPos < 0 )yPos = 0;
465
466 wxTreeItemIdValue cookie1;
467
468 childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie1 );
469 while ( childID.IsOk( ) )
470 {
471 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
472 child->SetXPos( xPos );
473 child->SetYPos( yPos );
474
475 // AlbumBaseType baseType = GetAlbumTreeCtrl( )->GetItemType( childID );
476
477 double childTop = ( GetHeight( ) - maxHeight ) / 2
478 - GetTopContentMargin( )
479 - GetBottomContentMargin( );
480
481 if ( childTop < 0 )childTop = 0;
482
483
484 if ( child->IsNodeType( AT_Stamp ) )
485 {
486 Design::AlignmentModeType stampAlign = GetAlignmentModeType( );
487
488 Stamp* stamp = ( Stamp* ) child;
489 Design::StampNamePosType stampNamePosition = stamp->GetStampNamePositionType( );
490 // std::cout << stamp->GetNameFrame( )->GetString( ) << "\n";
491
492 if ( maxTitleHeight < stamp->GetNameFrame( )->GetHeight( ) )
493 {
494 std::cout << "Error in Row::UpdatePosition "
495 << stamp->GetNameFrame( )->GetString( ) << " "
496 << " maxTitleHeight" << maxTitleHeight
497 << " Height " << stamp->GetNameFrame( )->GetHeight( )
498 << "\n";
499 while ( 0 )
500 {
501 };
502 }
503 if ( stampNamePosition == AT_StampNamePositionTop )
504 {
505 // align stamp top
506 if ( stampAlign == AlignTop )
507 {
508 child->SetYPos( yPos
509 + childTop
510 + maxTitleHeight
511 - stamp->GetNameFrame( )->GetHeight( ) );
512 }
513 // align stamp bottom
514 else if ( stampAlign == AlignBottom )
515 {
516 child->SetYPos( yPos
517 + childTop
518 + maxHeight
519 - stamp->GetHeight( ) );
520 }
521 // align stamp middle
522 else if ( stampAlign == AlignMiddle )
523 {
524 child->SetYPos( yPos
525 + childTop
526 + maxTitleHeight
527 - stamp->GetNameFrame( )->GetHeight( )
528 + maxFrameHeight / 2
529 - stamp->GetBorderFrame( )->GetHeight( ) / 2 );
530 }
531 }
532 else if ( stampNamePosition == AT_StampNamePositionBottom )
533 {
534 // align stamp top
535 if ( stampAlign == AlignTop )
536 {
537 child->SetYPos( yPos
538 + childTop );
539 }
540 // align stamp bottom
541 if ( stampAlign == AlignBottom )
542 {
543 child->SetYPos( yPos
544 + childTop
545 + maxFrameHeight
546 - stamp->GetBorderFrame( )->GetHeight( ) );
547 }
548 // align stamp middle
549 if ( stampAlign == AlignMiddle )
550 {
551 child->SetYPos( yPos
552 + childTop
553 + maxFrameHeight / 2
554 - stamp->GetBorderFrame( )->GetHeight( ) / 2 );
555 }
556 }
557 }
558 else
559 {
560 child->SetYPos( yPos + childTop );
561 }
562
563 child->UpdatePositions( );
564
565 xPos += child->GetWidth( ) + spacing;
566
567 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie1 );
568 }
569 ValidateNode( );
570 }

574 void Row::UpdateSizes( )
575 {
576 // Set the height and width of each child column
577 // Stamps have fixed height and width
578 wxTreeItemIdValue cookie;
579 wxTreeItemId parentID = GetTreeItemId( );
580 LayoutBase* parent = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( parentID );
581 int nbrChildren = parent->GetNbrChildren( );
582
583 wxTreeItemId childID = GetAlbumTreeCtrl( )->GetFirstChild( parentID, cookie );
584 while ( childID.IsOk( ) )
585 {
586 LayoutBase* child = ( LayoutBase* ) GetAlbumTreeCtrl( )->GetItemNode( childID );
587
588 if ( child->IsNodeType( AT_Col ) ) // i.e., stamps are fixed don't try to change them
589 {
590 child->SetHeight( GetHeight( )
591 - GetTopContentMargin( )
592 - GetBottomContentMargin( ) );
593 child->SetWidth( child->GetMinWidth( )
594 + ( GetWidth( ) - GetMinWidth( ) ) / ( 2 * nbrChildren ) // instead of just min width, make it a little bigger
595 - GetLeftContentMargin( )
596 - GetRightContentMargin( ) );
597 }
598 child->UpdateSizes( );
599 childID = GetAlbumTreeCtrl( )->GetNextChild( parentID, cookie );
600 }
