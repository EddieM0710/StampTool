256void RowDetailsPanel::SetAlignmentMode( )
257{
258 Design::AlignmentModeType mode = m_row->GetAlignmentModeType( );
259 if ( mode == Design::AlignTop )
260 {
261 m_alignTopButton->SetValue( true );
262 }
263 else if ( mode == Design::AlignMiddle )
264 {
265 m_alignMiddleButton->SetValue( true );
266 }
267 else if ( mode == Design::AlignBottom )
268 {
269 m_alignBottomButton->SetValue( true );
270 }
271 else
272 {
273 m_alignDefaultButton->SetValue( true );;
274 }
