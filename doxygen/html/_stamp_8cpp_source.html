117 {
118 Design::NodeStatus status = GetNodeStatus( );
119 if ( status != Design::AT_FATAL )
120 {
121 SetClientDimensions( dc, x + GetXPos( ), y + GetYPos( ), GetMinWidth( ), GetMinHeight( ) );
122
123 //Draw the Stamp frame
124 double xInnerPos = x + GetXPos( );
125 double yInnerPos = y + GetYPos( );
126
127 dc.SetPen( *wxBLACK_PEN );
128 m_borderFrame.Draw( dc, xInnerPos, yInnerPos );
129
130 double xImagePos = xInnerPos + m_stampImageFrame.GetXPos( );
131 double yImagePos = yInnerPos + m_stampImageFrame.GetYPos( );

133 if ( yImagePos < 0.0 )yImagePos = .01;
134
135 wxString filename = GetStampImageFilename( );
136

146 {
147 //Draw the stamp image

159 // Draw missing image frame transparent

169 // std::cout << GetNameFrame( )->GetString( ) << "\n";

655 void Stamp::SetFixedParms( )
656 {
657
658 // everything is basedfrom the actual stamp m_actualStampFrame
659 m_borderFrame.SetHeight( m_actualStampFrame.GetHeight( ) + GetSelvageHeight( ) + GetMountAllowanceHeight( ) );
660 m_borderFrame.SetWidth( m_actualStampFrame.GetWidth( ) + GetSelvageWidth( ) + GetMountAllowanceWidth( ) );
661
662 // The width of the stamp object is the m_borderFrame width + margin
663 SetWidth( m_borderFrame.GetWidth( ) + 2 * GetStampMargin( ) );
664
665 // The NameFrame vals are determined by the outer stamp object width, the text and the font.
666 // The string may be divided between several lines affecting height.
667 // Note the name is centered in the outside stamp object
668 UpdateString( GetNameFrame( ), GetWidth( ) );
669
670 // the height of the stamp object is the border frame height + border + title height
671 if ( GetShowTitle( ) )
672 {
673 SetHeight( GetNameFrame( )->GetHeight( ) + m_borderFrame.GetHeight( ) + 1 );
674 }
675 else
676 {
677 SetHeight( m_borderFrame.GetHeight( ) + 1 );
678 }
679
680 SetMinWidth( GetWidth( ) );
681 SetMinHeight( GetHeight( ) );
682
683
684
685 // the NbrFrame vals are determined by stamp frame width, the text and the font
686 // note the nbr is centered in the m_borderFrame
687 UpdateString( GetNbrFrame( ), m_borderFrame.GetWidth( ) );
688
689 m_stampImageFrame.SetWidth( m_actualStampFrame.GetWidth( ) * ImagePercentOfActual );
690 m_stampImageFrame.SetHeight( m_actualStampFrame.GetHeight( ) * ImagePercentOfActual );
691
692 }

713 void Stamp::UpdatePositions( )
714 {
715 // the xPos and yPos are offsets from the containing frame
716 //the m_borderFrame is offset by the margin.
717 m_borderFrame.SetXPos( GetStampMargin( ) );
718
719 // the name frame has 0 x offset but is centered in its frame
720 GetNameFrame( )->SetXPos( 0 );
721
722 // the number g=frame is centered within the border frame
723 GetNbrFrame( )->SetXPos( m_borderFrame.GetXPos( ) + ( m_borderFrame.GetWidth( ) - m_nbrFrame->GetWidth( ) ) / 2 );
724
725 // the image is horizontally centered within the m_borderFrame.
726 m_stampImageFrame.SetXPos( m_borderFrame.GetXPos( ) + ( m_borderFrame.GetWidth( ) - m_stampImageFrame.GetWidth( ) ) / 2 );
727 if ( GetShowTitle( ) )
728 {
729 if ( GetStampNamePositionType( ) == AT_StampNamePositionBottom )
730 {
731 //if the title is on the bottom then the border frame is on top
732 m_borderFrame.SetYPos( 0 );
733 // and the name is just below the m_borderFrame
734 m_nameFrame->SetYPos( m_borderFrame.GetHeight( ) + 1 );
735 }
736 else
737 {
738 // if the name is on top then no offset for the name frame
739 m_nameFrame->SetYPos( 0 );
740 // and the border is just under that
741
742 m_borderFrame.SetYPos( GetNameFrame( )->GetHeight( ) + 1 );
743 }
744 }
745 else
746 {
747 m_borderFrame.SetYPos( 0 );
748 }
749 // center the image vertically within the m_borderFrame
750 // note the image offsets from the m_borderFrame
751 m_stampImageFrame.SetYPos( m_borderFrame.GetYPos( ) + ( m_borderFrame.GetHeight( ) - m_stampImageFrame.GetHeight( ) - 4 ) / 2 );
752 m_nbrFrame->SetYPos( m_stampImageFrame.GetYPos( ) + m_stampImageFrame.GetHeight( ) );
753
754 ValidateNode( );
755 }

759 NodeStatus Stamp::ValidateNode( )
760 {
761 NodeStatus status = AT_OK;
762 wxString filename = GetStampImageFilename( );
763 wxString str;// = GetProject( )->GetImageFullPath( filename );
764 wxImage image = GetProject( )->GetImage( filename );
765 if ( !image.IsOk( ) )
766 {
767 std::cout << "\nInvalid Stamp Image. " << filename << "\n";
768 str = wxString::Format( "Invalid Stamp Image.\n" );
769 GetErrorArray( )->Add( str );
770 //SetError( AT_InvalidImage, AT_WARNING );
771 status = AT_WARNING;
772 }
773
774 if ( m_actualStampFrame.GetHeight( ) <= 0.01 )
775 {
776 str = wxString::Format( "Invalid Stamp Height.\n" );
777 GetErrorArray( )->Add( str );
778 //SetError( AT_InvalidHeight, AT_FATAL );
779 status = AT_FATAL;
780 }
781
782 if ( m_actualStampFrame.GetWidth( ) <= 0.01 )
783 {
784 str = wxString::Format( "Invalid Stamp Width.\n" );
785 GetErrorArray( )->Add( str );
786 status = AT_FATAL;
787 }
788 m_nodeValid = status;
789 wxTreeItemId id = GetTreeItemId( );
790 if ( id.IsOk( ) )
791 {
792 if ( status == AT_FATAL )
793 {
794 GetAlbumTreeCtrl( )->SetItemBackgroundColour( id, *wxRED );
795 std::cout << GetAlbumTreeCtrl( )->GetItemText( id ) << " Fatal\n";
796 }
797 else if ( status == AT_WARNING )
798 {
799 GetAlbumTreeCtrl( )->SetItemBackgroundColour( id, *wxYELLOW );
800 std::cout << GetAlbumTreeCtrl( )->GetItemText( id ) << " Warning\n";
801 }
802 else
803 {
804 GetAlbumTreeCtrl( )->SetItemBackgroundColour( id, *wxWHITE );
805 //std::cout << GetAlbumTreeCtrl( )->GetItemText( id ) << " OK\n";
806 }
807 }
808 return status;
809 }
