681void StampDetailsPanel::SetCatCodes()
682{
683 m_catCodesString = m_stamp->GetCatalogCodes();
684 if (m_catCodesString.IsEmpty())
685 {
686 wxTreeItemId m_designTreeID = GetAlbumTreeCtrl()->GetSelection();
687 if (m_designTreeID.IsOk())
688 {
689 DesignTreeItemData *data = (DesignTreeItemData *)GetAlbumTreeCtrl()->GetItemData(m_designTreeID);
690
691 Utils::StampLink *link = data->GetStampLink();
692 if (link)
693 {
694 wxTreeItemId catTreeID = link->GetCatTreeID();
695 wxXmlNode *node = GetCatalogTreeCtrl()->GetItemNode(catTreeID);
696
697 if (node)
698 {
699 // Catalog::CatalogVolume* catalogVolume = GetCatalogVolume( );
700 Catalog::Entry stamp(node);
701 m_catCodesString = stamp.GetCatalogCodes();
702 m_stamp->SetCatalogCodes(m_catCodesString);
703 }
704 }
705 }
706 }
707 if (!m_catCodesString.IsEmpty())
708 {
709 Catalog::CatalogCode catCodeArray(m_catCodesString);
710 m_catCodesStringArray = catCodeArray.GetCodes();
711 wxString cat = m_stamp->GetCatalog();
712 wxString preferredID = catCodeArray.GetPreferredCatalogCode(cat);
713 m_catCode->ChangeValue(m_catCodesString);
714 m_catNbr->Clear();
715 m_catNbr->Append(m_catCodesStringArray);
716 int ndx = m_catNbr->FindString(preferredID);
717 m_stamp->SetNbrString(preferredID);
718 m_catNbr->SetSelection(ndx);
719 return;
720 }
721 else
722 {
723 wxString str = m_stamp->GetNbrString();
724 m_catNbr->Clear();
725 m_catNbr->Append(str);
726 m_catNbr->SetSelection(0);
727 return;
728 }
729 m_catCode->ChangeValue("");
730}
