304 void StampMountData::FixUpHyphenID( wxXmlNode* node, wxString level )
305 {
306 level += " ";
307 wxXmlNode* child = node->GetChildren( );
308 int count = 0;
309 while ( child )
310 {
311 wxXmlNodeType type = child->GetType( );
312 wxString nodeName = child->GetName( );
313
314 // std::cout << level << nodeName << " ";
315
316 wxXmlAttribute* attr = child->GetAttributes( );
317 while ( attr )
318 {
319 wxString name = attr->GetName( );
320 wxString val = attr->GetValue( );
321 // std::cout << level << " " << name << ":" << val << "\n";
322 attr = attr->GetNext( );
323 }
324 wxString val = "";
325 val = child->GetAttribute( Catalog::MountDataTypeNames[ MT_ID ], val );
326 int pos = val.Find( "-" );
327 wxString prefix = "Sn:US_";
328 if ( pos != wxNOT_FOUND )
329 {
330 wxString startStr = val.Mid( 0, pos );
331 int startLen = startStr.Length( );
332 wxString lastStr = val.Mid( pos + 1 );
333 int lastLen = lastStr.Length( );
334 wxString last = startStr.Mid( 0, startLen - lastLen ) + lastStr;
335 // std::cout << val << " " << startStr << " " << last << "\n";
336 val = prefix + startStr;
337 // std::cout << val << " i1 ";
338 count++;
339 Utils::SetAttrStr( child, Catalog::MountDataTypeNames[ MT_ID ], val );
340 int begin;
341 startStr.ToInt( &begin );
342 begin++;
343 int end;
344 lastStr.ToInt( &end );
345 wxXmlNode* prevNode = child;
346 wxString year = "";
347 year = child->GetAttribute( Catalog::MountDataTypeNames[ MT_Year ], year );
348
349 //wxString id = "";
350 //id = child->GetAttribute( Catalog::MountDataTypeNames[ MT_ID ], id );
351
352 wxString name = "";

357
358 wxString mult = "";
359 mult = child->GetAttribute( Catalog::MountDataTypeNames[ MT_MultipleMount ], mult );
360
361 wxString block = "";
362 block = child->GetAttribute( Catalog::MountDataTypeNames[ MT_BlockMount ], block );
363
364 wxString sheet = "";
365 sheet = child->GetAttribute( Catalog::MountDataTypeNames[ MT_SheetMount ], sheet );
366
367 for ( int i = begin; i <= end; i++ )
368 {
369
370 wxXmlNode* newNode = new wxXmlNode( 0, wxXML_ELEMENT_NODE, "Item" );
371 newNode->AddAttribute( Catalog::MountDataTypeNames[ MT_Year ], year );
372 wxString id;
373 id = id.Format( "%s%d", prefix, i );
374 // std::cout << id << " i2 ";
375 count++;
376 newNode->AddAttribute( Catalog::MountDataTypeNames[ MT_ID ], id );
377
378 newNode->AddAttribute( Catalog::MountDataTypeNames[ MT_Name ], name );
379 newNode->AddAttribute( Catalog::MountDataTypeNames[ MT_SingleMount ], single );
380 newNode->AddAttribute( Catalog::MountDataTypeNames[ MT_MultipleMount ], mult );
381 newNode->AddAttribute( Catalog::MountDataTypeNames[ MT_BlockMount ], block );
382 newNode->AddAttribute( Catalog::MountDataTypeNames[ MT_SheetMount ], sheet );
383 node->InsertChildAfter( newNode, prevNode );
384 prevNode = newNode;
385 }
386 }
387 else
388 {
389 wxString id = prefix + val;
390 // std::cout << id << " i3 ";
391 count++;
392 Utils::SetAttrStr( child, Catalog::MountDataTypeNames[ MT_ID ], id );
393 }
394
395 // std::cout << "\n";
396 if ( child->GetChildren( ) )
397 {
398 FixUpHyphenID( child, level );
399 }
400 if ( count > 15 )
401 {
402 // std::cout << "\n";

404 }
