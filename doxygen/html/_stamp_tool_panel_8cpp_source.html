188void StampToolPanel::OnNotebookPageChanged( wxNotebookEvent& event )
189{
190 int sel = m_notebook->GetSelection( );
191
192 wxWindow* page = m_notebook->GetPage( sel );
193
194 // TESTFIX
195 if ( page == m_stampDescriptionPanel )
196 {
197 m_stampDescriptionPanel->Show();
198 m_generateListPanel->Hide();
199 m_albumAlbumPanel->Hide();
200 }
201 else if ( page == m_generateListPanel )
202 {
203 m_stampDescriptionPanel->Hide();
204 m_generateListPanel->Show();
205 m_albumAlbumPanel->Hide();
206 }
207 else if ( page == m_albumAlbumPanel )
208 {
209 m_stampDescriptionPanel->Hide();
210 m_generateListPanel->Hide();
211 m_albumAlbumPanel->Show();
212 }
213 // ENDTESTFIX
214
215 wxString catCode = "";
216 if ( page != m_stampDescriptionPanel )
217 {
218 wxTreeItemId selId = m_catalogPanel->GetCatalogTree( )->GetSelection( );
219 m_catalogPanel->GetCatalogTree( )->GetItemNode( selId );
220 Catalog::CatalogBaseType type = m_catalogPanel->GetCatalogTree( )->GetItemType( selId );
221 if ( type == Catalog::NT_Entry )
222 {
223 wxXmlNode* node = m_catalogPanel->GetCatalogTree( )->GetItemNode( selId );
224 Catalog::Entry* entry = new Catalog::Entry( node );
225 catCode = entry->GetID( );
226
227 }
228 m_catalogPanel->GetCatalogTree( )->SetStates( true );
229 m_catalogPanel->GetCatalogTree( )->LoadCatalogTree( );
230 if ( !catCode.IsEmpty( ) )
231 {
232 GetCatalogTreeCtrl( )->SelectStamp( catCode );
233 }
234 }
235 else //if ( page == m_stampDescriptionPanel )
236 {
237 wxTreeItemId selId = m_catalogPanel->GetCatalogTree( )->GetSelection( );
238 m_catalogPanel->GetCatalogTree( )->GetItemNode( selId );
239 Catalog::CatalogBaseType type = m_catalogPanel->GetCatalogTree( )->GetItemType( selId );
240 if ( type == Catalog::NT_Entry )
241 {
242 wxXmlNode* node = m_catalogPanel->GetCatalogTree( )->GetItemNode( selId );
243 Catalog::Entry* entry = new Catalog::Entry( node );
244 catCode = entry->GetID( );
245 }
246 m_catalogPanel->GetCatalogTree( )->SetStates( false );
247 m_catalogPanel->GetCatalogTree( )->LoadCatalogTree( );
248 if ( !catCode.IsEmpty( ) )
249 {
250 GetCatalogTreeCtrl( )->SelectStamp( catCode );
251 }
252 }
253 Layout();

259void StampToolPanel::OnPDF( wxString filename )
260{
261 wxFileName fileName( filename );
262
263 wxPrintData printData;
264 printData.SetOrientation( wxPORTRAIT );
265 printData.SetPaperId( wxPAPER_A4 );
266 printData.SetFilename( fileName.GetFullPath( ) );
267
268
269 wxFileType* fileType = wxTheMimeTypesManager->GetFileTypeFromExtension( wxS( "pdf" ) );
270 if ( fileType != NULL )
271 {
272 wxString cmd = fileType->GetOpenCommand( fileName.GetFullPath( ) );
273 if ( !cmd.IsEmpty( ) )
274 {
275 wxExecute( cmd );
276 }
277 delete fileType;
278 }
