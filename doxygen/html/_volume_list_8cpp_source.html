47 wxXmlNode* VolumeList::AddChild( wxXmlNode* child )
48 {
49 wxString label;
50
51 wxString nodeName = child->GetName( );
52 Utils::TOCBaseType nodeType = FindTOCBaseType( nodeName );
53 if ( nodeType == Utils::TOC_Volume )
54 {
55
56 // make the volume
57 wxString filename = Utils::GetAttrStr( child, "FileName" );
58 Utils::Volume* vol = NewVolumeInstance( filename );
59
60 vol->Load( );
61
62 wxString volName = vol->GetName( );
63 Utils::SetAttrStr( child, "VolumeName", volName );
64
65 // Add it to the catalog List
66 Insert( volName, vol );
67
68 }
69
70 // now do it all again for this entrys children
71 wxXmlNode* grandChild = child->GetChildren( );
72 while ( grandChild )
73 {
74 // start adding child elements to it.
75 AddChild( grandChild );
76 grandChild = grandChild->GetNext( );
77 }
78 return child;
79 }
