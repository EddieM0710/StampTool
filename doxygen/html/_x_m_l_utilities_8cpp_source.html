74 wxXmlNode* XMLIterator::First( )
75 {
76 m_firstDone = true;
77 // get the first child and save the location
78 m_currElement = m_parent->GetChildren( );
79
80 // if the currElement is null there is noting to iterate
81 if ( !m_currElement )
82 {
83 return m_currElement;
84 }
85 if ( m_currElement->GetChildren( ) )
86 {
87 // if the currElement has no children return it and set its sibling as
88 // curr
89 wxXmlNode* retElement = m_currElement;
90
91 m_currElement = retElement->GetNext( );
92 return retElement;
93 }
94 else
95 {
96 // if it has children setup a next level iterator and return the
97 // currElement
98 wxXmlNode* retElement = m_currElement;
99 if ( m_decend )
100 {
101 m_iterator = new XMLIterator( m_currElement );
102 }
103 m_currElement = retElement->GetNext( );
104 return retElement;
105 }
106 }
107 wxXmlNode* XMLIterator::Next( )
108 {
109 if ( !m_firstDone )
110 {
111 return First( );
112 }
113 // if there is an iterator defined then drill down to process
114 if ( m_iterator )
115 {
116 wxXmlNode* nextElement = m_iterator->Next( );
117 if ( nextElement )
118 {
119 return nextElement;
120 }
121 // if there is no next element then we are thru processing
122 // this elements children; go to the next one.
123
124 if ( m_iterator )
125 {
126 delete m_iterator;
127 m_iterator = 0;
128 }
129 }
130
131 if ( !m_currElement )
132 {
133 return m_currElement;
134 }
135
136 if ( m_currElement->GetChildren( ) )
137 {
138 wxXmlNode* retElement = m_currElement;
139 m_currElement = retElement->GetNext( );
140 return retElement;
141 }
142 else
143 {
144 wxXmlNode* retElement = m_currElement;
145 if ( m_decend )
146 {
147 m_iterator = new XMLIterator( m_currElement );
148 }
149 m_currElement = retElement->GetNext( );
150 return retElement;
151 }
152 }
